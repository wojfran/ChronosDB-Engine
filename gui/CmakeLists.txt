find_package(Qt6 COMPONENTS Widgets Charts Concurrent REQUIRED)

set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC OFF)
set(CMAKE_AUTORCC OFF)

add_executable(chronosdb_gui
    src/main.cpp
    src/MainWindow.cpp
    include/gui/MainWindow.h
    src/LogController.cpp
    include/gui/LogController.h
    src/SignalListController.cpp
    include/gui/SignalListController.h
    src/ChartController.cpp
    include/gui/ChartController.h
)

target_include_directories(chronosdb_gui PUBLIC include)

target_link_libraries(chronosdb_gui
    PRIVATE
        Qt6::Widgets
        Qt6::Charts
        Qt6::Concurrent
        chronosdb_core
        common
)
